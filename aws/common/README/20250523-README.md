# 🌐 Frontend EC2 Instance Setup Guide (Manual SSL Certificate)

이 문서는 Terraform과 shell script(`user_data`)를 활용해 프론트엔드 EC2 인스턴스를 자동 구성하고, 기존 인스턴스에서 복사한 SSL 인증서를 통해 HTTPS 서비스를 구성하는 가이드입니다.

---

## 🛠️ 1. Terraform `user_data` 구성

**경로 예시**: `modules/instance/frontend-user-data.sh`

```bash
#!/bin/bash
set -eux

# 필수 패키지 설치
apt-get update -y
apt-get install -y curl gnupg2 ca-certificates lsb-release nginx build-essential

# Node.js v23 설치
curl -fsSL https://deb.nodesource.com/setup_23.x | bash -
apt-get install -y nodejs

# pnpm 설치
npm install -g pnpm

# .env 파일 생성
cat > /home/ubuntu/.env <<EOF
${env_content}
EOF

chown ubuntu:ubuntu /home/ubuntu/.env
chmod 600 /home/ubuntu/.env

# Nginx 설정
cat > /etc/nginx/sites-available/tebutebu.com <<EOF
server {
    server_name tebutebu.com;

    location /api/ {
        proxy_set_header Host \$host;
        proxy_set_header X-Forwarded-Proto \$scheme;
        proxy_set_header X-Forwarded-Host \$host;
        proxy_set_header X-Forwarded-Port \$server_port;
        proxy_set_header X-Real-IP \$remote_addr;
        proxy_pass http://${backend_ip}:8080;
    }

    location /oauth2/ {
        proxy_set_header Host \$host;
        proxy_set_header X-Forwarded-Proto \$scheme;
        proxy_set_header X-Forwarded-Port \$server_port;
        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-Host \$host;
        proxy_pass http://${backend_ip}:8080;
    }

    location / {
        proxy_pass http://localhost:3000/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade \$http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host \$host;
        proxy_cache_bypass \$http_upgrade;
    }

    listen 443 ssl;
    ssl_certificate /etc/letsencrypt/live/tebutebu.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/tebutebu.com/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
}

server {
    if (\$host = tebutebu.com) {
        return 301 https://\$host\$request_uri;
    }

    listen 80;
    server_name tebutebu.com;
    return 404;
}
EOF

# Nginx 적용
ln -sf /etc/nginx/sites-available/tebutebu.com /etc/nginx/sites-enabled/default
systemctl restart nginx
systemctl enable nginx
